```typescript
import cache from '@/utils/cache';

async function handler(ctx) {
    const { dept, path } = ctx.req.param();
    const baseUrl = `https://${dept}.hebei.gov.cn`;
    const listUrl = `${baseUrl}/${path}/`;
    
    const response = await got({ url: listUrl });

        items.map((item) =>
            cache.tryGet(item.link, async () => {

                item.description = content('.TRS_Editor, .content, .article-content, .main-content').html();
```

### 二、解决 IP 封锁的 3 个可行方案（优先级排序）

这是你最大的瓶颈（6 个省级重要厅局被封），需要优先解决：

#### 方案 1：在 RSSHub 容器中加住宅/隧道代理出口（最推荐 ✅）

在 `universal_proxy.mjs` 的 `fetch()` 里加一层代理出口：

```javascript
import { SocksProxyAgent } from 'socks-proxy-agent';
import { HttpsProxyAgent } from 'https-proxy-agent';

// 封锁 IP 段列表
const BLOCKED_IPS = new Set([
    '121.29.48.87', '121.29.49.195', '124.239.213.29',
    '121.29.48.192', '36.144.10.42', '218.11.12.4'
]);

// 住宅代理（付费服务，如快代理/芝麻代理/IPIPGO 的国内隧道代理）
const RESIDENTIAL_PROXY = 'http://user:pass@tunnel.kuaidaili.com:15818';
// 或 SOCKS5: 'socks5://user:pass@xxx:1080'

async function fetchWithFallback(targetUrl, headers) {
    // 先 DNS 解析看是否在封锁段
    const hostname = new URL(targetUrl).hostname;
    const resolved = await dns.promises.resolve4(hostname).catch(() => []);
    const needProxy = resolved.some(ip => BLOCKED_IPS.has(ip));
    
    const fetchOptions = { headers, signal: AbortSignal.timeout(25000) };
    
    if (needProxy) {
        // 走住宅代理出口
        const agent = new HttpsProxyAgent(RESIDENTIAL_PROXY);
        fetchOptions.agent = agent;  // node-fetch / undici 支持
    }
    
    return fetch(targetUrl, fetchOptions);
}
```

**推荐代理服务**（国内住宅代理，政府站通常只允许国内 IP）：
- 快代理隧道代理（HTTP隧道，每请求换IP）：约 30-50元/月起
- 芝麻代理：按请求计费
- IPIPGO：按流量计费
- 或者最便宜的方案：**用家里宽带搭一个 frp/wireguard 隧道**，把家里的 IP 暴露给服务器用

#### 方案 2：家用宽带 WireGuard 隧道（免费但需要设备）

如果你家有一台常开设备（路由器/NAS/旧电脑）：

```bash
# 家里设备上：
wg genkey | tee privatekey | wg pubkey > publickey
# 配 WireGuard，把家里设备作为出口

# 服务器上：
# 通过 WireGuard 隧道把特定流量路由到家里 IP 出去
ip route add 121.29.48.0/24 via <wg_peer_ip>
ip route add 36.144.10.0/24 via <wg_peer_ip>
```

然后 Node 代理直接通过 WireGuard 接口发起请求，出口 IP 就是你家宽带的 IP（政府站一般不封家用宽带段）。

#### 方案 3：多云出口（成本最高）

在阿里云/华为云再开一台最低配 ECS（河北区域），专门做代理中继。政府站通常对本省云厂商 IP 段友好。

### 四、优化现有 fetcher 的具体建议

#### 4.1 请求指纹升级（Chrome 120 → 136+）

你的 UA 是 Chrome/120，现在已经是 2026 年，政府站 WAF 可能会逐步标记老版本。建议升级：

```python
headers = {
    "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/136.0.0.0 Safari/537.36",
    "Sec-Ch-Ua": '"Chromium";v="136", "Google Chrome";v="136", "Not.A/Brand";v="99"',
    "Sec-Ch-Ua-Platform": '"Windows"',
    "Sec-Ch-Ua-Mobile": "?0",
    # 其余不变
}
```

#### 4.2 增加 Cookie 持久化

部分站点（如 kjj）第一次访问返回空内容是因为需要先完成一次 302 redirect + set-cookie。建议在 Node 代理里加 cookie jar：

```javascript
// universal_proxy.mjs 升级
import { CookieJar } from 'tough-cookie';

const cookieJar = new CookieJar();

async function fetchWithCookies(targetUrl, headers) {
    const cookies = await cookieJar.getCookieString(targetUrl);
    if (cookies) headers['Cookie'] = cookies;
    
    const resp = await fetch(targetUrl, { headers, redirect: 'manual' });
    
    // 保存 Set-Cookie
    const setCookies = resp.headers.raw()['set-cookie'] || [];
    for (const sc of setCookies) {
        await cookieJar.setCookie(sc, targetUrl);
    }
    
    // 处理 302 重定向
    if (resp.status === 302) {
        const location = resp.headers.get('location');
        return fetchWithCookies(new URL(location, targetUrl).href, headers);
    }
    
    return resp;
}
```

#### 4.3 针对 JS 渲染站点的 API 拦截

对于 `kjj.qhd.gov.cn` 这类列表页靠 JS 动态加载的站点，更好的做法是**拦截 XHR/fetch 请求**而不是等渲染：

```python
# 在 Playwright 里拦截 API 请求
async def intercept_api(page, url):
    api_responses = []
    
    async def handle_response(response):
        if 'api' in response.url or 'list' in response.url.lower():
            try:
                json_data = await response.json()
                api_responses.append({'url': response.url, 'data': json_data})
            except:
                pass
    
    page.on('response', handle_response)
    await page.goto(url, wait_until='networkidle')
    return api_responses
```

#### 4.4 增加请求间隔随机化

```python
import random
import asyncio

async def rate_limited_fetch(url):
    # 同域名请求间加 2-5 秒随机延迟
    delay = random.uniform(2.0, 5.0)
    await asyncio.sleep(delay)
    return await fetch(url)
```
